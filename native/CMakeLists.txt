cmake_minimum_required(VERSION 3.31.6)
project(DeviceIdChanger LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "DeviceIdChanger requires Android arm64-v8a")
endif()
if(NOT ANDROID_PLATFORM_LEVEL EQUAL 29)
    message(FATAL_ERROR "Build with ANDROID_PLATFORM=android-29")
endif()
if(NOT ANDROID_STL STREQUAL "none")
    message(FATAL_ERROR "Build with ANDROID_STL=none; pinned standalone libcxx is provided")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(cmake/Dependencies.cmake)
set(cxx_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(cxx CONFIG REQUIRED)
link_libraries(cxx::cxx)

set(LSPLANT_STANDALONE ON CACHE BOOL "" FORCE)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DEX_BUILDER_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(dependencies/lsplant/lsplant/src/main/jni dependencies/lsplant-build EXCLUDE_FROM_ALL)

set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(dependencies/dobby dependencies/dobby-build EXCLUDE_FROM_ALL)

set(XdlRoot "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/xdl/xdl/src/main/cpp")
file(GLOB XdlSources CONFIGURE_DEPENDS "${XdlRoot}/*.c")
list(REMOVE_ITEM XdlSources "${XdlRoot}/xdl.c")
add_library(DeviceIdXdl STATIC ${XdlSources} Source/Xdl.c)
target_include_directories(DeviceIdXdl SYSTEM PRIVATE "${XdlRoot}" PUBLIC "${XdlRoot}/include")
target_compile_features(DeviceIdXdl PRIVATE c_std_17)
target_compile_options(DeviceIdXdl PRIVATE -fvisibility=hidden)
set_source_files_properties(Source/Xdl.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")

add_library(deviceidchanger SHARED Source/Module.cpp Source/Profile.cpp Source/Runtime.cpp)
set_target_properties(deviceidchanger PROPERTIES CXX_STANDARD 20 CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_include_directories(deviceidchanger SYSTEM PRIVATE dependencies/zygisk/module/jni dependencies/lsplant/lsplant/src/main/jni/include dependencies/dobby/include)
target_compile_options(deviceidchanger PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fno-exceptions -fno-rtti)
target_link_libraries(deviceidchanger PRIVATE lsplant_static dobby_static DeviceIdXdl log dl z)
target_link_options(deviceidchanger PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/Exports.map")
