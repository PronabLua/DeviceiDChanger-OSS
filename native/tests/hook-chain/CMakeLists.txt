cmake_minimum_required(VERSION 3.31.6)
project(HookChain LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Hook chain test requires Android arm64-v8a")
endif()
get_filename_component(NativeRoot "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(RepositoryRoot "${NativeRoot}/.." ABSOLUTE)
set(FirstDobbyLibrary "${NativeRoot}/build/dependencies/dobby-build/libdobby.a" CACHE FILEPATH "First hook engine static library")
set(VectorDobbySource "${RepositoryRoot}/build/research/override-dobby-vector" CACHE PATH "Exact Vector Dobby checkout")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${VectorDobbySource}" rev-parse HEAD OUTPUT_VARIABLE VectorCommit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT VectorCommit STREQUAL "05a09ac6807a6bb1726350e40ea4b127c1c79809")
    message(FATAL_ERROR "Vector Dobby source revision mismatch")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(cxx_DIR "${NativeRoot}/cmake")
find_package(cxx CONFIG REQUIRED)
link_libraries(cxx::cxx)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory("${VectorDobbySource}" "${CMAKE_CURRENT_BINARY_DIR}/vector-dobby" EXCLUDE_FROM_ALL)
add_library(FirstDobby STATIC IMPORTED)
set_target_properties(FirstDobby PROPERTIES IMPORTED_LOCATION "${FirstDobbyLibrary}")
add_library(chain_first SHARED First.cpp)
target_include_directories(chain_first SYSTEM PRIVATE "${NativeRoot}/dependencies/dobby/include")
target_link_libraries(chain_first PRIVATE FirstDobby log dl)
add_library(chain_second SHARED Second.cpp)
target_include_directories(chain_second SYSTEM PRIVATE "${VectorDobbySource}/include")
target_link_libraries(chain_second PRIVATE dobby_static log dl)
foreach(Target chain_first chain_second)
    target_compile_options(${Target} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fno-exceptions -fno-rtti)
    target_link_options(${Target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/Exports.map")
endforeach()
add_executable(hook_chain_test Main.cpp)
target_compile_options(hook_chain_test PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fno-exceptions -fno-rtti)
target_link_libraries(hook_chain_test PRIVATE dl)
target_link_options(hook_chain_test PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
